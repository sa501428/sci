cmake_minimum_required(VERSION 3.16)
project(sci_cpp VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(STRAW_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../hic-infra/straw/C++" CACHE PATH "Straw C++ source directory")
if(NOT EXISTS "${STRAW_SOURCE_DIR}/straw_v10.h")
  message(FATAL_ERROR "Set -DSTRAW_SOURCE_DIR=/path/to/straw/C++ (streaming reader required)")
endif()
find_package(CURL REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
find_package(zstd CONFIG QUIET)
if(TARGET zstd::libzstd_static)
  set(SCI_ZSTD zstd::libzstd_static)
elseif(TARGET zstd::libzstd_shared)
  set(SCI_ZSTD zstd::libzstd_shared)
elseif(TARGET zstd::libzstd)
  set(SCI_ZSTD zstd::libzstd)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(ZSTD REQUIRED IMPORTED_TARGET libzstd)
  set(SCI_ZSTD PkgConfig::ZSTD)
endif()
add_library(sci_straw STATIC "${STRAW_SOURCE_DIR}/straw.cpp" "${STRAW_SOURCE_DIR}/straw_v10.cpp"
  "${STRAW_SOURCE_DIR}/hbs.cpp" "${STRAW_SOURCE_DIR}/norm_vectors.cpp")
set_target_properties(sci_straw PROPERTIES CXX_STANDARD 14)
target_include_directories(sci_straw PUBLIC "${STRAW_SOURCE_DIR}")
target_link_libraries(sci_straw PUBLIC CURL::libcurl ZLIB::ZLIB Threads::Threads ${SCI_ZSTD})
add_library(sci_core cpp/core.cpp)
target_include_directories(sci_core PUBLIC cpp)
target_link_libraries(sci_core PUBLIC Threads::Threads)
target_compile_options(sci_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(sci_cpp cpp/main.cpp)
set_target_properties(sci_cpp PROPERTIES OUTPUT_NAME sci)
target_link_libraries(sci_cpp PRIVATE sci_core sci_straw)
target_compile_options(sci_cpp PRIVATE -Wall -Wextra -Wpedantic)
option(SCI_NATIVE "Optimize for the build machine CPU" OFF)
if(SCI_NATIVE)
  target_compile_options(sci_core PRIVATE -march=native)
endif()
include(CheckIPOSupported)
check_ipo_supported(RESULT SCI_IPO)
if(SCI_IPO)
  set_property(TARGET sci_core sci_cpp PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
install(TARGETS sci_cpp RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(sci_core_test tests/core.cpp)
  target_link_libraries(sci_core_test PRIVATE sci_core)
  target_compile_options(sci_core_test PRIVATE -UNDEBUG)
  add_test(NAME core COMMAND sci_core_test)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME hic_pipeline COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:sci_cpp>")
  set_tests_properties(core hic_pipeline PROPERTIES TIMEOUT 60)
endif()
